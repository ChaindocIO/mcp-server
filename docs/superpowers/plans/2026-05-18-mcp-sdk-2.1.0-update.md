# MCP server v0.2.0 — Chaindoc SDK 2.1.0 update — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Expand the Chaindoc MCP server from 35 tools (SDK 2.0.x surface) to 102 tools (SDK 2.1.0 full coverage) plus a verification-polling helper, hybrid binary-download adapter, and structured state-machine error refinement — without breaking any existing tool, resource, or prompt.

**Architecture:** Thin, typed 1:1 façade over `@chaindoc_io/server-sdk@^2.1.0`. Each MCP tool = one Zod input schema + one `client.run(...)` SDK call. Three new tool files cover the new nested SDK namespaces (`contracts.termination`, `contracts.paymentTerms`, `contracts.agreements`); existing module files grow in place. Downloads dispatch through a transport-aware adapter (disk on stdio, capped inline base64 on HTTP). Errors get a pattern-matched refinement layer that turns string-message `400`s into agent-actionable codes plus hints.

**Tech Stack:** TypeScript 5.6, Node 18.18+, `@modelcontextprotocol/sdk@^1.0.0`, `@chaindoc_io/server-sdk@^2.1.0`, `zod@^3.23.8`, `vitest@^2.1.0`, `pino@^9.5.0`, `express@^4.21.0`.

**Spec:** [docs/superpowers/specs/2026-05-18-mcp-sdk-2.1.0-update-design.md](../specs/2026-05-18-mcp-sdk-2.1.0-update-design.md).

```ts
// src/core/downloads.ts
import { mkdir, writeFile } from "node:fs/promises";
import { join } from "node:path";
import { randomUUID, createHash } from "node:crypto";
import { Buffer } from "node:buffer";

export type DownloadAdapterMode = "auto" | "disk" | "inline";

export interface DownloadAdapterConfig {
  mode: DownloadAdapterMode;
  dir: string;
  inlineMaxBytes: number;
}

export interface DownloadResult {
  data: ArrayBuffer;
  contentType: string;
  fileName?: string;
}

export interface DiskResult {
  kind: "disk";
  path: string;
  contentType: string;
  fileName?: string;
  sizeBytes: number;
  sha256: string;
}

export interface InlineResult {
  kind: "inline";
  base64: string;
  contentType: string;
  fileName?: string;
  sizeBytes: number;
  sha256: string;
}

const EXT_BY_CONTENT_TYPE: Record<string, string> = {
  "application/pdf": "pdf",
  "image/png": "png",
  "image/jpeg": "jpg",
  "image/gif": "gif",
  "image/webp": "webp",
  "image/svg+xml": "svg",
  "text/html": "html",
  "text/plain": "txt",
  "application/json": "json",
  "application/octet-stream": "bin",
};

function inferExt(contentType: string, suggested?: string): string {
  if (suggested) return suggested.replace(/^\./, "");
  const base = contentType.split(";")[0]?.trim().toLowerCase() ?? "";
  return EXT_BY_CONTENT_TYPE[base] ?? "bin";
}

export async function deliverDownload(
  cfg: DownloadAdapterConfig,
  res: DownloadResult,
  hint: { tool: string; suggestedExt?: string },
): Promise<DiskResult | InlineResult> {
  const bytes = new Uint8Array(res.data);
  const sizeBytes = bytes.byteLength;
  const sha256 = createHash("sha256").update(bytes).digest("hex");
  const mode = cfg.mode === "auto" ? "disk" : cfg.mode;

  if (mode === "inline") {
    if (sizeBytes > cfg.inlineMaxBytes) {
      throw Object.assign(
        new Error(
          `Download is ${sizeBytes} bytes; inline cap is ${cfg.inlineMaxBytes}. ` +
            `Raise CHAINDOC_INLINE_DOWNLOAD_MAX_BYTES or run the MCP server on stdio to land the file on disk.`,
        ),
        { status: 413, code: "payload_too_large" },
      );
    }
    const out: InlineResult = {
      kind: "inline",
      base64: Buffer.from(bytes).toString("base64"),
      contentType: res.contentType,
      sizeBytes,
      sha256,
    };
    if (res.fileName !== undefined) out.fileName = res.fileName;
    return out;
  }

  await mkdir(cfg.dir, { recursive: true, mode: 0o700 });
  const ext = inferExt(res.contentType, hint.suggestedExt);
  const fileName = res.fileName ?? `${hint.tool}-${randomUUID()}.${ext}`;
  const filePath = join(cfg.dir, fileName);
  await writeFile(filePath, bytes, { mode: 0o600 });
  const out: DiskResult = {
    kind: "disk",
    path: filePath,
    contentType: res.contentType,
    sizeBytes,
    sha256,
  };
  if (res.fileName !== undefined) out.fileName = res.fileName;
  return out;
}
```

Recognize the `code` field already attached by the download adapter and the `response.message` array shape from SDK doc §2.2. Replace `toChaindocMcpError` with:

```ts
const STATUS_TO_CODE: Record<number, ChaindocErrorCode> = {
  400: "invalid_argument",
  401: "unauthenticated",
  403: "forbidden",
  404: "not_found",
  409: "conflict",
  413: "payload_too_large",
  422: "unprocessable",
  429: "rate_limited",
};

interface SdkErrorShape {
  status?: number;
  statusCode?: number;
  message?: string;
  code?: string;
  requestId?: string;
  retryAfter?: number;
  body?: unknown;
  response?: { message?: unknown };
}

export function toChaindocMcpError(err: unknown): ChaindocMcpError {
  if (err instanceof NotImplementedError) {
    return { code: "not_implemented", message: err.message };
  }
  if (typeof err === "object" && err !== null) {
    const e = err as SdkErrorShape;
    const status = e.status ?? e.statusCode;
    if (typeof status === "number") {
      // Honor an explicit `code` from the source (e.g. the download adapter's `payload_too_large`).
      const baseCode =
        (typeof e.code === "string" && STATUS_TO_CODE[status] !== undefined
          ? (e.code as ChaindocErrorCode)
          : undefined) ??
        STATUS_TO_CODE[status] ??
        (status >= 500 ? "upstream_error" : "internal_error");
      const out: ChaindocMcpError = {
        code: baseCode,
        message: e.message ?? `Chaindoc API returned ${status}`,
        httpStatus: status,
      };
      if (typeof e.requestId === "string") out.requestId = e.requestId;
      if (typeof e.retryAfter === "number") out.retryAfter = e.retryAfter;
      // Lift validation-array messages out of details into `validation`.
      const responseMessage = e.response?.message;
      if (Array.isArray(responseMessage)) {
        out.validation = responseMessage;
      } else if (e.body !== undefined) {
        out.details = e.body;
      }
      return out;
    }
  }
  if (err instanceof Error) return { code: "internal_error", message: err.message };
  return { code: "internal_error", message: String(err) };
}
```

```ts
    "Block until a document version's blockchain anchor is verified (or timeoutMs elapses). " +
    "Use after chaindoc_documents_create with status='published' and before chaindoc_signatures_create_request. " +
    "Polls chaindoc_documents_get_verification_status every intervalMs.",
  inputSchema: z.object({
    versionId: z.string().uuid(),
    timeoutMs: z.number().int().positive().max(120_000).default(60_000),
    intervalMs: z.number().int().positive().min(500).max(10_000).default(2_000),
  }),
  handler: async (args, { client }) => {
    const deadline = Date.now() + args.timeoutMs;
    while (Date.now() < deadline) {
      const status = await client.run("documents.wait_for_verification.poll", (sdk) =>
        sdk.documents.getVerificationStatus(args.versionId),
      );
      if ((status as { verified?: boolean }).verified === true) {
        return { verified: true, status };
      }
      await new Promise((r) => setTimeout(r, args.intervalMs));
    }
    return {
      verified: false,
      timedOut: true,
      message:
        "Version not verified within timeout. The blockchain anchor may still be in flight; retry the same poll later.",
    };
  },
});
```
